static void drawExplosion(float x, float y, float r, float phase) {
    DrawCircleV({x, y}, r, Color{255, 160, 30, 255});
    DrawCircleV({x, y}, r * 0.62f, Color{255, 255, 170, 255});
    for (int i = 0; i < 10; ++i) {
        float a = phase + i * 0.6283185f;
        float r0 = r * 0.7f + (i % 3) * 2.0f;
        float r1 = r * (1.5f + (i % 4) * 0.15f);
        DrawLineEx({x + std::cos(a) * r0, y + std::sin(a) * r0},
                   {x + std::cos(a) * r1, y + std::sin(a) * r1},
                   3, Color{255, 120, 20, 255});
    }
}

static void drawCombatScreen(const CombatView& v, Texture2D playerTex, Texture2D enemyTex) {
    int W = GetScreenWidth(), H = GetScreenHeight();
    const Color NAVY = {8, 8, 60, 255};
    const Color PANEL = {14, 14, 92, 255};
    const Color GOLD = {224, 192, 24, 255};
    const Color TXT = {236, 236, 240, 255};
    const Color DIM = {168, 168, 178, 255};
    ClearBackground(NAVY);

    int topH = (int)(H * 0.155f);
    int midH = (int)(H * 0.60f);
    int botY = topH + midH;
    int botH = H - botY;

    const Unit& A = gMission.units[v.atk];
    const Unit& D = gMission.units[v.def];
    const UnitType& TA = gTypes[A.type];
    const UnitType& TD = gTypes[D.type];

    // ----- top: names + EXP
    DrawRectangle(0, 0, W, topH, NAVY);
    int nameSize = (int)(topH * 0.34f);
    DrawText(TA.name.c_str(), (int)(W * 0.06f), (int)(topH * 0.30f), nameSize, TXT);
    DrawText(TD.name.c_str(), (int)(W * 0.72f), (int)(topH * 0.30f), nameSize, TXT);
    DrawText("EXP", (int)(W * 0.29f), (int)(topH * 0.34f), (int)(topH * 0.26f), TXT);
    DrawText("EXP", (int)(W * 0.64f), (int)(topH * 0.34f), (int)(topH * 0.26f), TXT);
    auto expBox = [&](float x, int val) {
        float s = topH * 0.66f;
        float y = topH * 0.15f;
        DrawRectangleLinesEx({x, y, s, s}, 3, GOLD);
        DrawText(TextFormat("%d", val), (int)(x + s * 0.34f), (int)(y + s * 0.16f),
                 (int)(s * 0.72f), GOLD);
    };
    expBox((float)(W * 0.36f), A.rank + 1);
    expBox((float)(W * 0.56f), D.rank + 1);

    // ----- middle: terrain panel(s)
    DrawRectangle(0, topH, W, midH, NAVY);
    auto drawPanel = [&](int idx, Rectangle dst) {
        if (gCombatPanels.empty() || !gCombatPanels[idx].id) return;
        Texture2D& t = gCombatPanels[idx];
        DrawTexturePro(t, {0, 0, (float)t.width, (float)t.height}, dst, {0, 0}, 0, WHITE);
    };
    if (v.panelA == v.panelB) {
        drawPanel(v.panelA, {(float)0, (float)topH, (float)W, (float)midH});
    } else {
        drawPanel(v.panelA, {(float)0, (float)topH, (float)(W / 2), (float)midH});
        drawPanel(v.panelB, {(float)(W / 2), (float)topH, (float)(W / 2), (float)midH});
        DrawRectangle(W / 2 - 2, topH, 4, midH, NAVY);
    }

    // ----- armies: 8 vehicles each, losses remove sprites
    float t = v.t;
    int nA = (v.dmgAtk > 0 && t > 1.30f) ? (v.hpA1 + 1) : (v.hpA0 + 1);
    int nD = (t > 0.55f) ? (v.hpD1 + 1) : (v.hpD0 + 1);
    nA = std::clamp(nA, 0, 8);
    nD = std::clamp(nD, 0, 8);

    static const float FORM[8][2] = {
        {-2.4f, -1.4f}, {0.0f, -1.9f}, {2.2f, -1.1f}, {-2.8f, 0.4f},
        {-0.4f, -0.1f}, {2.6f, 0.7f},  {-1.9f, 2.0f},  {0.9f, 2.1f}
    };
    float us = midH * 0.135f;
    float cy = topH + midH * 0.50f;
    float ax = W * 0.23f, dx = W * 0.77f;

    auto drawArmy = [&](float cx, int count, int type, Texture2D& tex, Color tint) {
        int idx = 128 + type;
        Rectangle src{(float)((idx % 32) * 32), (float)((idx / 32) * 32), 32, 32};
        for (int i = 0; i < count; ++i) {
            float ox = FORM[i][0] * us * 0.95f;
            float oy = FORM[i][1] * us * 0.80f;
            DrawTexturePro(tex, src,
                           {cx + ox - us / 2, cy + oy - us / 2, us, us},
                           {0, 0}, 0, tint);
        }
    };

    Color tintA = (v.dmgAtk > 0 && t > 1.30f && t < 1.65f) ? Color{255, 140, 140, 255} : WHITE;
    Color tintD = (t > 0.55f && t < 0.95f) ? Color{255, 140, 140, 255} : WHITE;
    drawArmy(ax, nA, A.type, playerTex, tintA);
    drawArmy(dx, nD, D.type, enemyTex, tintD);

    if (t > 0.42f && t < 1.00f) {
        for (int i = 0; i < 3; ++i) {
            float ph = t * 7.0f + i * 1.7f;
            drawExplosion(dx + std::sin(ph) * us * 2.0f,
                          cy + std::cos(ph * 1.3f) * us * 1.6f,
                          us * (0.55f + 0.15f * std::sin(ph * 2.0f)), ph);
        }
    }
    if (v.dmgAtk > 0 && t > 1.00f && t < 1.45f) {
        for (int i = 0; i < 2; ++i) {
            float ph = t * 7.0f + i * 2.1f;
            drawExplosion(ax + std::sin(ph) * us * 1.8f,
                          cy + std::cos(ph) * us * 1.4f, us * 0.5f, ph);
        }
    }
    if (t > 0.28f && t < 0.50f)
        DrawCircleV({ax + us * 3.0f, cy}, us * 0.35f, YELLOW);

    if (v.dmgDef > 0 && t > 0.55f)
        DrawText(TextFormat("-%d", v.dmgDef), (int)(W * 0.62f), topH + 12,
                 (int)(H * 0.045f), Color{255, 90, 90, 255});
    if (v.dmgAtk > 0 && t > 1.30f)
        DrawText(TextFormat("-%d", v.dmgAtk), (int)(W * 0.10f), topH + 12,
                 (int)(H * 0.045f), ORANGE);

    // ----- bottom: attack / defense / terrain
    DrawRectangle(0, botY, W, botH, NAVY);
    auto statBox = [&](float x, float w, const char* l1, const char* l2) {
        float h = botH * 0.70f;
        float y = botY + botH * 0.15f;
        DrawRectangleRec({x, y, w, h}, PANEL);
        DrawRectangleLinesEx({x, y, w, h}, 3, GOLD);
        int fs = (int)(botH * 0.22f);
        DrawText(l1, (int)(x + w * 0.06f), (int)(y + h * 0.16f), fs, TXT);
        DrawText(l2, (int)(x + w * 0.06f), (int)(y + h * 0.52f), fs, TXT);
    };
    statBox(W * 0.03f, W * 0.25f, TextFormat("ATK %d", v.atkStrength),
            TextFormat("DEF %d", v.defStrength));
    statBox(W * 0.72f, W * 0.25f, TextFormat("ATK %d", v.ctrStrength),
            TextFormat("DEF %d", v.ctrDefStrength));

    DrawText("TERRAIN", (int)(W * 0.43f), (int)(botY + botH * 0.14f),
             (int)(botH * 0.22f), TXT);
    DrawText(TextFormat("%d%%", gTerrain[std::clamp(v.terrainA, 0, 15)].def),
             (int)(W * 0.39f), (int)(botY + botH * 0.54f), (int)(botH * 0.22f), DIM);
    DrawText(TextFormat("%d%%", gTerrain[std::clamp(v.terrainB, 0, 15)].def),
             (int)(W * 0.54f), (int)(botY + botH * 0.54f), (int)(botH * 0.22f), DIM);

    DrawText("SPACE / click", W / 2 - 70, H - (int)(botH * 0.18f),
             (int)(botH * 0.15f), DIM);
}
